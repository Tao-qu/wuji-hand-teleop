cmake_minimum_required(VERSION 3.16)
project(gmsl_camera)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GSTREAMER REQUIRED IMPORTED_TARGET gstreamer-app-1.0 gstreamer-video-1.0)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs)
find_package(yaml-cpp REQUIRED)
find_package(Threads REQUIRED)
add_executable(gmsl_camera_node src/capture_backend.cpp src/gmsl_camera_node.cpp)
target_compile_features(gmsl_camera_node PRIVATE cxx_std_17)
target_compile_options(gmsl_camera_node PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(gmsl_camera_node PRIVATE include ${OpenCV_INCLUDE_DIRS})
ament_target_dependencies(gmsl_camera_node rclcpp sensor_msgs diagnostic_msgs)
target_link_libraries(gmsl_camera_node PkgConfig::GSTREAMER ${OpenCV_LIBS} yaml-cpp Threads::Threads)
install(TARGETS gmsl_camera_node DESTINATION lib/${PROJECT_NAME})
ament_package()
